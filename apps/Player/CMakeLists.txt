cmake_minimum_required(VERSION 3.20)
project(SimpleMidiProject CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(SIMPLEMIDI_CLI_ONLY "Build SimpleMidi without its graphical interface" OFF)

if(NOT SIMPLEMIDI_CLI_ONLY)
    # GLFW — disable docs/tests/examples, we only want the library
    set(GLFW_BUILD_DOCS     OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS    OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    add_subdirectory(${CMAKE_SOURCE_DIR}/external/glfw ${CMAKE_BINARY_DIR}/external/glfw)    

    # ImGui — compiled as a static library
    set(IMGUI_DIR ${CMAKE_SOURCE_DIR}/external/imgui)
    add_library(imgui STATIC
        ${IMGUI_DIR}/imgui.cpp
        ${IMGUI_DIR}/imgui_draw.cpp
        ${IMGUI_DIR}/imgui_tables.cpp
        ${IMGUI_DIR}/imgui_widgets.cpp
        ${IMGUI_DIR}/backends/imgui_impl_glfw.cpp
        ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp
    )
    target_include_directories(imgui PUBLIC
        ${IMGUI_DIR}
        ${IMGUI_DIR}/backends
    )
    target_link_libraries(imgui PUBLIC glfw)
endif()

# --- The Main Application (includes audio.cpp) ---
add_executable(SimpleMidi
    SimpleMidi.cpp
    inst-piano.cpp
    inst-mt32.cpp
    audio.cpp
)


# Linux (ALSA) dependencies for MIDI handling
if(UNIX AND NOT APPLE)
    find_package(ALSA REQUIRED)
    if(ALSA_FOUND)
        target_include_directories(SimpleMidi PRIVATE ${ALSA_INCLUDE_DIRS})
        target_link_libraries(SimpleMidi PUBLIC ${ALSA_LIBRARY})
    endif()
endif()

target_link_libraries(SimpleMidi PRIVATE midilib)

# 2. Path directly to the mt32emu library folder inside munt
set(MT32EMU_DIR "${CMAKE_SOURCE_DIR}/instruments/munt/mt32emu")

# 3. Check if mt32emu exists, compile it, and link it
if(EXISTS "${MT32EMU_DIR}/CMakeLists.txt")
    message(STATUS "Found mt32emu library. Adding subdirectory to compile it.")
    
    # Optional: Configure mt32emu settings before adding it
    set(mt32emu_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(libmt32emu_SHARED OFF CACHE BOOL "Build mt32emu as static library" FORCE)
    
    # This now compiles ONLY the core mt32emu library CMake project
    add_subdirectory(${CMAKE_SOURCE_DIR}/instruments/munt/mt32emu ${CMAKE_BINARY_DIR}/instruments/munt/mt32emu)

    # Link against the target generated by mt32emu's CMake
    target_link_libraries(SimpleMidi PRIVATE mt32emu)

    # Conditionally include your MT-32 integration source file
    target_sources(SimpleMidi PRIVATE inst-mt32.cpp)
    
    target_compile_definitions(SimpleMidi PRIVATE HAS_MT32EMU=1)
    message(STATUS "mt32emu linked and inst-mt32.cpp included.")
else()
    message(STATUS "mt32emu not found at ${MT32EMU_DIR}. Skipping compilation.")
    target_compile_definitions(SimpleMidi PRIVATE HAS_MT32EMU=0)
endif()

if(SIMPLEMIDI_CLI_ONLY)
    target_compile_definitions(SimpleMidi PRIVATE SIMPLEMIDI_CLI_ONLY)
else()
    target_link_libraries(SimpleMidi PRIVATE imgui)
    if(APPLE)
        target_link_libraries(SimpleMidi PRIVATE "-framework OpenGL")
    elseif(WIN32)
        target_link_libraries(SimpleMidi PRIVATE opengl32)
    else()
        find_package(OpenGL REQUIRED)
        target_link_libraries(SimpleMidi PRIVATE OpenGL::GL)
    endif()
endif()
